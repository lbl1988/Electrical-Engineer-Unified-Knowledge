# RAG 四维检索

> 输入自然语言查询，调用四维检索 API（向量 + 关键词 + 图谱 + 标准溯源），返回融合排序结果。
> 架构与数据流见 [RAG 检索使用指南](20-standards/rag-usage-guide.md)。

<script>
function ragSearch() {
  const q = document.getElementById('rag-query').value.trim();
  if (!q) { alert('请输入查询'); return; }

  const status = document.getElementById('rag-status');
  const results = document.getElementById('rag-results');
  status.textContent = '检索中...';
  results.innerHTML = '';

  const apiBase = window.KB_API_BASE || 'http://127.0.0.1:8000';
  const url = `${apiBase}/search?query=${encodeURIComponent(q)}&top_k=10`;

  fetch(url)
    .then(r => {
      if (!r.ok) throw new Error(`HTTP ${r.status}: ${r.statusText}`);
      return r.json();
    })
    .then(data => {
      status.textContent = `返回 ${data.count} 条结果（查询：${data.query}）`;
      if (!data.results || data.results.length === 0) {
        results.innerHTML = '<p style="color:#999;">无匹配结果</p>';
        return;
      }
      results.innerHTML = data.results.map((it, i) => {
        const tag = `<span class="rag-tag ${it.source}">${it.source}</span>`;
        const title = it.title || '(无标题)';
        const layer = it.layer || '';
        const score = it.score || 0;
        const text = (it.text || '').replace(/</g, '&lt;').replace(/>/g, '&gt;');
        const link = it.source_path
          ? `<a class="rag-link" href="${it.source_path}">查看原文</a>`
          : '';
        return `
          <div class="rag-card">
            <h3>${i + 1}. ${title} <span style="font-size:12px;color:#999;">[${it.entry_id}]</span></h3>
            <div class="rag-meta">
              ${tag}
              <span>层级：${layer}</span> ｜
              <span>得分：${score}</span>
              ${it.section ? '｜ <span>节：' + it.section + '</span>' : ''}
              ${link ? ' ｜ ' + link : ''}
            </div>
            <div class="rag-text">${text}</div>
          </div>
        `;
      }).join('');
    })
    .catch(err => {
      status.textContent = '检索失败：' + err.message;
      status.style.color = '#c62828';
    });
}

document.getElementById('rag-query').addEventListener('keydown', function(e) {
  if (e.key === 'Enter') ragSearch();
});
</script>

## 检索维度说明

| 维度 | 说明 | 标签 |
|---|---|---|
| 语义向量 | Embedding 余弦相似度，概念匹配 | `vector` |
| 关键词 | BM25 全文匹配，精确术语 | `keyword` |
| 知识图谱 | BFS 多跳，关联发现 | `graph` |
| 标准溯源 | 规范条款锚定 | `standard` |

## 示例查询
